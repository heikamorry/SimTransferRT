cmake_minimum_required(VERSION 3.21)
project(SimTransferRT VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SIM_WITH_QT "Build pure-code Qt GUI, Qt CLI and application tests" ON)
option(SIM_BUILD_TESTS "Build tests" ON)
option(SIM_SANITIZERS "ASan + UBSan on GCC/Clang" OFF)
option(SIM_DEPLOY_QT "Install Qt runtime dependencies on Windows" ON)
find_package(Threads REQUIRED)
include(GNUInstallDirs)
if(MSVC)
    add_compile_options(/utf-8 /W4 /permissive-)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
    if(SIM_SANITIZERS)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
endif()
if(SIM_WITH_QT)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Widgets)
    qt_standard_project_setup()
    set(CMAKE_AUTOUIC OFF) # Entire interface is C++, no .ui or QML.
    set(CMAKE_AUTORCC ON)
endif()
add_subdirectory(src)
if(SIM_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
