cmake_minimum_required(VERSION 3.21)
project(SimTransferRT VERSION 0.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SIM_WITH_QT "Build pure-code Qt GUI, Qt CLI and application tests" ON)
option(SIM_MODEL_PREVIEW "Include local model conversion and static preview in the Qt GUI" ON)
option(SIM_BUILD_TESTS "Build tests" ON)
option(SIM_SANITIZERS "ASan + UBSan on GCC/Clang" OFF)
option(SIM_DEPLOY_QT "Install Qt runtime dependencies on Windows" ON)
find_package(Threads REQUIRED)
include(GNUInstallDirs)
if(MSVC)
    add_compile_options(/utf-8 /W4 /permissive-)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
    if(SIM_SANITIZERS)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
endif()
if(SIM_WITH_QT)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets)
    if(SIM_MODEL_PREVIEW)
        include(src/model/Dependencies.cmake)
    endif()
    qt_standard_project_setup()
    set(CMAKE_AUTOUIC OFF) # Entire interface is C++, no .ui or QML.
    set(CMAKE_AUTORCC ON)
endif()
option(SIM_WITH_ASIO "Build the pinned Boost.Asio V2 transport (required by the asio GUI)" ON)
if(SIM_WITH_QT AND NOT SIM_WITH_ASIO)
    message(FATAL_ERROR "This branch is Asio-only: the GUI requires SIM_WITH_ASIO=ON. Run scripts/setup-asio-deps.ps1 first.")
endif()
if(SIM_WITH_ASIO)
    include(cmake/AsioDependencies.cmake)
endif()
add_subdirectory(src)
if(WIN32 AND SIM_WITH_QT)
    add_library(SimSetupActions SHARED EXCLUDE_FROM_ALL installer/SetupActions.cpp)
    target_link_libraries(SimSetupActions PRIVATE msi shell32 ole32 advapi32)
    set_target_properties(SimSetupActions PROPERTIES AUTOMOC OFF AUTORCC OFF
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(SIM_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
